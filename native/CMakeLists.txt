cmake_minimum_required(VERSION 3.20)
project(bf6_godot_binding LANGUAGES CXX)
set(BF6_CORE_PACKAGE "" CACHE PATH "Matched reader package containing include/, lib/ and bin/")
set(BF6_CORE_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../core" CACHE PATH "BF6_High_Poly_Core checkout (the core submodule)")
foreach(required include/bf6_core.h lib/bf6_core.lib bin/bf6_core.dll reader-manifest.json)
    if(NOT EXISTS "${BF6_CORE_PACKAGE}/${required}")
        message(FATAL_ERROR "Missing reader package file: ${BF6_CORE_PACKAGE}/${required}")
    endif()
endforeach()
add_library(bf6_oodle SHARED src/bf6_oodle.cpp)
target_compile_features(bf6_oodle PRIVATE cxx_std_17)
target_include_directories(bf6_oodle PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${BF6_CORE_PACKAGE}/include")
target_link_libraries(bf6_oodle PRIVATE "${BF6_CORE_PACKAGE}/lib/bf6_core.lib")
set_target_properties(bf6_oodle PROPERTIES OUTPUT_NAME "bf6_oodle.windows.x86_64")
if(MSVC)
    target_compile_options(bf6_oodle PRIVATE /W3 /EHsc)
endif()
add_custom_command(TARGET bf6_oodle POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${BF6_CORE_PACKAGE}/bin/bf6_core.dll" "$<TARGET_FILE_DIR:bf6_oodle>/bf6_core.dll")
